def load_and_preprocess_images(image_dir, mask_dir, target_size=(256, 256), num_augmented_images=800):
    # Define the same seed to ensure same transformation on image and mask
    seed = 42

    # Image data generator
    image_datagen = ImageDataGenerator(
        rotation_range=20,
        width_shift_range=0.1,
        height_shift_range=0.1,
        shear_range=0.1,
        zoom_range=0.1,
        horizontal_flip=True,
        fill_mode='nearest'
    )

    # Mask data generator — no interpolation to preserve binary values
    mask_datagen = ImageDataGenerator(
        rotation_range=20,
        width_shift_range=0.1,
        height_shift_range=0.1,
        shear_range=0.1,
        zoom_range=0.1,
        horizontal_flip=True,
        fill_mode='nearest'
    )

    image_filenames = [f for f in os.listdir(image_dir)
                       if f.lower().endswith(('.png', '.jpg', '.jpeg')) and
                          os.path.exists(os.path.join(mask_dir, f))]

    print(f"Found {len(image_filenames)} matched image-mask pairs.")

    augmented_images = []
    augmented_masks = []

    while len(augmented_images) < num_augmented_images:
        filename = random.choice(image_filenames)

        # Load and resize image
        img_path = os.path.join(image_dir, filename)
        mask_path = os.path.join(mask_dir, filename)

        img = load_img(img_path, target_size=target_size)
        mask = load_img(mask_path, target_size=target_size, color_mode="grayscale")

        x_img = img_to_array(img)
        x_mask = img_to_array(mask)

        # Expand dimensions to make shape (1, h, w, c)
        x_img = np.expand_dims(x_img, axis=0)
        x_mask = np.expand_dims(x_mask, axis=0)

        # Create generators
        img_gen = image_datagen.flow(x_img, batch_size=1, seed=seed)
        mask_gen = mask_datagen.flow(x_mask, batch_size=1, seed=seed)

        # Generate one pair
        aug_img = next(img_gen)[0].astype(np.uint8)
        aug_mask = next(mask_gen)[0]

        # Ensure binary mask (e.g., values exactly 0 or 255, or 0 and 1)
        aug_mask = (aug_mask > 127).astype(np.uint8) * 255

        augmented_images.append(aug_img)
        augmented_masks.append(aug_mask)

    augmented_images = np.array(augmented_images, dtype=np.uint8)
    augmented_masks = np.array(augmented_masks, dtype=np.uint8)

    print(f"Generated {len(augmented_images)} image-mask pairs.")

    return augmented_images, augmented_masks

def custom_unet(input_shape=(256, 256, 3)): # TO MATCH WHAT THE GIVEN PROCESSING FUNCTION CONVERTS THE IMGS TO, CHANGE FROM 128,128,1 to 256,256,3
    inputs = Input(input_shape)

    # Encoder (build from scratch)
    e1 = Conv2D(64, 3, activation='relu', padding='same')(inputs)
    e1 = Conv2D(64, 3, activation='relu', padding='same')(e1)
    e1 = Dropout(0.1)(e1)
    p1 = MaxPooling2D(pool_size=(2, 2))(e1)

    # e2
    e2 = Conv2D(128, 3, activation='relu', padding='same')(p1)
    e2 = Conv2D(128, 3, activation='relu', padding='same')(e2)
    e2 = Dropout(0.1)(e2)
    p2 = MaxPooling2D(pool_size=(2, 2))(e2)

    # e3
    e3 = Conv2D(256, 3, activation='relu', padding='same')(p2)
    e3 = Conv2D(256, 3, activation='relu', padding='same')(e3)
    e3 = Dropout(0.1)(e3)
    p3 = MaxPooling2D(pool_size=(2, 2))(e3)

    # e4
    e4 = Conv2D(512, 3, activation='relu', padding='same')(p3)
    e4 = Conv2D(512, 3, activation='relu', padding='same')(e4)
    e4 = Dropout(0.1)(e4)
    p4 = MaxPooling2D(pool_size=(2, 2))(e4)

    # e5
    e5 = Conv2D(1024, 3, activation='relu', padding='same')(p4)
    e5 = Conv2D(1024, 3, activation='relu', padding='same')(e5)
    e5 = Dropout(0.1)(e5)
    # no maxpooling


    # Decoder (upsampling and concatenation with skip connections)
    d1 = UpSampling2D()(e5)
    d1 = Concatenate()([d1, e4])
    d1 = Conv2D(512, 3, activation='relu', padding='same')(d1)
    d1 = Conv2D(512, 3, activation='relu', padding='same')(d1)
    d1 = Dropout(0.1)(d1)

    # d2
    d2 = UpSampling2D()(d1)
    d2 = Concatenate()([d2, e3])
    d2 = Conv2D(256, 3, activation='relu', padding='same')(d2)
    d2 = Conv2D(256, 3, activation='relu', padding='same')(d2)
    d2 = Dropout(0.1)(d2)

    # d3
    d3 = UpSampling2D()(d2)
    d3 = Concatenate()([d3, e2])
    d3 = Conv2D(128, 3, activation='relu', padding='same')(d3)
    d3 = Conv2D(128, 3, activation='relu', padding='same')(d3)
    d3 = Dropout(0.1)(d3)

    # d4
    d4 = UpSampling2D()(d3)
    d4 = Concatenate()([d4, e1])
    d4 = Conv2D(64, 3, activation='relu', padding='same')(d4)
    d4 = Conv2D(64, 3, activation='relu', padding='same')(d4)
    d4 = Dropout(0.1)(d4)

    # Final output layer
    output = Conv2D(1, 1, activation='sigmoid')(d4)

    # Build and return model
    model = Model(inputs, output)
    return model

model = custom_unet()

    initial_lrate = 0.0001

def calculate_mean_iou(y_true, y_pred):
  yt0 = tf.cast(y_true[:,:,:,0], tf.float32)
  yp0 = tf.cast(y_pred[:,:,0] > 0.5, tf.float32)
  intersection = tf.reduce_sum(tf.cast(tf.logical_and(tf.equal(yt0,1), tf.equal(yp0,1)), tf.float32))
  union = tf.reduce_sum(tf.cast(tf.add(yt0,yp0), tf.float32))
  iou = intersection / tf.maximum(union, 1e-10)
  return iou
